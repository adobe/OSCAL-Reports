⚠️  IMPORTANT: The application generates default credentials during
    the Docker build. You MUST change these after first login!

To extract the default credentials:
  kubectl exec -it deployment/{{ include "oscal-report-generator.fullname" . }} -- cat /app/credentials.txt

Or from Docker image directly:
  docker pull {{ .Values.image.repository }}:{{ .Values.image.tag | default .Chart.AppVersion }}
  docker create --name temp-oscal {{ .Values.image.repository }}:{{ .Values.image.tag | default .Chart.AppVersion }}
  docker cp temp-oscal:/app/credentials.txt ./credentials.txt
  cat credentials.txt
  docker rm temp-oscal

     export NODE_PORT={{ .Values.service.nodePort | default "auto-assigned" }}

     export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "oscal-report-generator.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
     echo "http://$SERVICE_IP:{{ .Values.service.port }}"

  ⏳ NOTE: It may take a few minutes for the LoadBalancer IP to be available.
{{- else if contains "ClusterIP" .Values.service.type }}
  🔗 Access the application by running:
     export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "oscal-report-generator.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
     echo "http://127.0.0.1:8080"
     kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:{{ .Values.app.port }}
{{- end }}
